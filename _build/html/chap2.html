sig_2 = np.zeros_like(t)

window_square = np.zeros_like(t)
window_square[np.abs(t - 0.5) < 0.5] = 1.0
window_square = np.roll(window_square, N // 2)

window_gaussian = np.exp(-0.5 * ((t - 0.5) / (tau / 8)) ** 2)
window_gaussian = np.roll(window_gaussian, N // 2)

conv_1 = scipy.signal.convolve(sig_1, window_square, mode = 'full') * dt
conv_2 = scipy.signal.convolve(sig_2, window_square, mode = 'full') * dt
conv_3 = scipy.signal.convolve(sig_3, window_gaussian, mode = 'full') * dt

    sig = np.zeros(N)
    sig[np.abs(m * t) < 0.5] = 1.0
    sig /= sig.sum() * dt
    return sig

M = 9
sigs = np.zeros((M, N))

sos = scipy.signal.butter(3, fb, 'low', fs = 1 / (t[1] - t[0]), output = 'sos')
